lemma Client_auth_injective:
  "All a b t #i.
     Commit(a, b, t) @ i
   ==>
       ( (Ex #j. Running(b, a, t) @ j
             & j < i
             & not (Ex a2 b2 #i2. Commit(a2, b2, t) @ i2
                                & not (#i2 = #i))) )
     | (Ex X #r. LtkReveal(X) @ r & Honest(X) @ i & r < i)"
