lemma agree_keys_inj:
  "All a b t #i.
     Commit(a, b, <'keys', t>) @ i
   ==>
       ( (Ex #j. Running(b, a, <'keys', t>) @ j
             & j < i
             & not (Ex a2 b2 #i2. Commit(a2, b2, <'keys', t>) @ i2
                                & not (#i2 = #i))) )
     | (Ex X #r. LtkReveal(X) @ r & Honest(X) @ i)"
